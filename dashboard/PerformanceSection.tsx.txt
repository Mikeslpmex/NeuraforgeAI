import React, { useState } from 'react';
import { HardwareNode } from '../types';
import { 
  Coins, 
  Smartphone, 
  Wifi, 
  Users, 
  BarChart3, 
  Copy, 
  Check, 
  Zap, 
  Info, 
  Cpu, 
  TrendingUp, 
  Laptop, 
  Volume2, 
  HelpCircle 
} from 'lucide-react';

interface PerformanceSectionProps {
  totalFgc: number;
  nodes: HardwareNode[];
  subAffiliatesCount: number;
  referralLink: string;
  miningRatePerSecond: number;
}

export default function PerformanceSection({
  totalFgc,
  nodes,
  subAffiliatesCount,
  referralLink,
  miningRatePerSecond
}: PerformanceSectionProps) {
  const [copied, setCopied] = useState(false);
  const [chartTimeframe, setChartTimeframe] = useState<'day' | 'week'>('day');

  // Calculates totals
  const totalPower = nodes.length > 0
    ? Math.round(nodes.reduce((acc, curr) => acc + (curr.status === 'online' ? curr.powerContribution : 0), 0) / nodes.length)
    : 0;
    
  const totalFgcPerHour = nodes.reduce((acc, curr) => acc + (curr.status === 'online' ? curr.fgcPerHour : 0), 0);

  // Expose device summaries nicely
  const mobCount = nodes.filter(n => n.type === 'mobile').length;
  const speakerCount = nodes.filter(n => n.type === 'smartspeaker').length;
  const routerCount = nodes.filter(n => n.type === 'router').length;
  const laptopCount = nodes.filter(n => n.type === 'laptop').length;

  const handleCopy = () => {
    navigator.clipboard.writeText(referralLink);
    setCopied(true);
    setTimeout(() => setCopied(false), 2000);
  };

  // SVG Chart Data Generator
  // We'll generate dynamic paths for the modern SVG line graph to represent FGC generation trends over a day or week.
  const getChartPoints = () => {
    if (chartTimeframe === 'day') {
      // 12 points for day
      return [
        { label: '00:00', fgc: 92 },
        { label: '02:00', fgc: 104 },
        { label: '04:00', fgc: 110 },
        { label: '08:00', fgc: 115 },
        { label: '10:00', fgc: 122 },
        { label: '12:00', fgc: totalFgc * 0.88 },
        { label: '14:00', fgc: totalFgc * 0.92 },
        { label: '16:00', fgc: totalFgc * 0.95 },
        { label: '18:00', fgc: totalFgc * 0.97 },
        { label: '20:00', fgc: totalFgc * 0.99 },
        { label: 'Ahora', fgc: totalFgc },
      ];
    } else {
      // 7 points for week
      return [
        { label: 'Lun', fgc: totalFgc - 84 },
        { label: 'Mar', fgc: totalFgc - 71 },
        { label: 'Mié', fgc: totalFgc - 55 },
        { label: 'Jue', fgc: totalFgc - 41 },
        { label: 'Vie', fgc: totalFgc - 28 },
        { label: 'Sáb', fgc: totalFgc - 12 },
        { label: 'Dom', fgc: totalFgc },
      ];
    }
  };

  const chartPoints = getChartPoints();
  const maxVal = Math.max(...chartPoints.map(p => p.fgc), 1);
  const minVal = Math.min(...chartPoints.map(p => p.fgc), 0);
  
  // Calculate SVG line path
  const width = 500;
  const height = 150;
  const padding = 30;
  
  const chartWidth = width - padding * 2;
  const chartHeight = height - padding * 2;
  
  const pointsString = chartPoints.map((p, index) => {
    const x = padding + (index / (chartPoints.length - 1)) * chartWidth;
    // inverse y-axis for SVG coordinate system
    const ratio = maxVal === minVal ? 0.5 : (p.fgc - minVal) / (maxVal - minVal);
    const y = padding + chartHeight - ratio * chartHeight;
    return `${x},${y}`;
  }).join(' ');

  // Create solid filled path for area under line
  const areaPointsString = chartPoints.length > 0 
    ? `${padding},${height - padding} ${pointsString} ${width - padding},${height - padding}`
    : '';

  return (
    <div className="flex flex-col gap-6">
      
      {/* 2x2 Grid of polished hardware and coin metrics */}
      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
        
        {/* Metric 1: Coins Generated */}
        <div className="relative group overflow-hidden bg-gradient-to-br from-black/50 to-purple-950/10 border border-purple-500/10 rounded-2xl p-5 hover:border-purple-500/30 transition-all duration-300">
          <div className="absolute top-2 right-2 w-8 h-8 rounded-lg bg-purple-500/10 flex items-center justify-center text-purple-400">
            <Coins className="w-4.5 h-4.5" />
          </div>
          <p className="text-[10px] font-mono font-bold tracking-widest text-[#00d2d3] uppercase">
            Forgecoins Generados
          </p>
          <div className="mt-2.5 flex items-baseline gap-1.5">
            <span className="text-3xl font-display font-extrabold tracking-tight text-white select-all">
              {totalFgc.toFixed(4)}
            </span>
            <span className="text-xs font-mono font-bold text-cyan-300">FGC</span>
          </div>
          <div className="mt-2 flex items-center justify-between text-[11px] font-mono text-gray-400">
            <span>≈ {(totalFgc * 0.00000825).toFixed(8)} BTC</span>
            <span className="text-emerald-400 flex items-center gap-0.5">
              <Zap className="w-3 h-3 text-emerald-400 inline" />
              +{(totalFgcPerHour / 3600).toFixed(5)} FGC/s
            </span>
          </div>
          <div className="w-full h-[2px] bg-purple-500/20 absolute bottom-0 left-0 hover:bg-[#00d2d3]/50 transition-colors" />
        </div>

        {/* Metric 2: Connected Nodes/Hardware */}
        <div className="relative group overflow-hidden bg-gradient-to-br from-black/50 to-cyan-950/10 border border-cyan-500/10 rounded-2xl p-5 hover:border-cyan-500/30 transition-all duration-300">
          <div className="absolute top-2 right-2 w-8 h-8 rounded-lg bg-cyan-500/10 flex items-center justify-center text-cyan-400">
            <Cpu className="w-4.5 h-4.5" />
          </div>
          <p className="text-[10px] font-mono font-bold tracking-widest text-purple-300 uppercase">
            Nodos Donados / Activos
          </p>
          <div className="mt-2.5 flex items-baseline gap-1.5">
            <span className="text-3xl font-display font-extrabold tracking-tight text-white">
              {nodes.length} Nodos
            </span>
          </div>
          <div className="mt-2 flex items-center gap-1.5 text-[11px] font-mono text-gray-400">
            <span className="flex items-center gap-0.5 bg-cyan-950/50 px-1.5 py-0.5 border border-cyan-500/20 rounded">
              <Smartphone className="w-3 h-3 text-cyan-400" /> {mobCount} Móv
            </span>
            <span className="flex items-center gap-0.5 bg-cyan-950/50 px-1.5 py-0.5 border border-cyan-500/20 rounded">
              <Volume2 className="w-3 h-3 text-cyan-400" /> {speakerCount} Smart
            </span>
            <span className="flex items-center gap-0.5 bg-cyan-950/50 px-1.5 py-0.5 border border-cyan-500/20 rounded">
              <Laptop className="w-3 h-3 text-cyan-400" /> {laptopCount + routerCount} Eq
            </span>
          </div>
          <div className="w-full h-[2px] bg-[#00d2d3]/20 absolute bottom-0 left-0" />
        </div>

        {/* Metric 3: Affiliated Network Bots */}
        <div className="relative group overflow-hidden bg-gradient-to-br from-black/50 to-amber-950/10 border border-amber-500/10 rounded-2xl p-5 hover:border-amber-500/30 transition-all duration-300">
          <div className="absolute top-2 right-2 w-8 h-8 rounded-lg bg-amber-500/10 flex items-center justify-center text-amber-400">
            <Users className="w-4.5 h-4.5" />
          </div>
          <p className="text-[10px] font-mono font-bold tracking-widest text-amber-400 uppercase">
            Bots Sub-Afiliados
          </p>
          <div className="mt-2.5 flex items-baseline gap-1.5">
            <span className="text-3xl font-display font-extrabold tracking-tight text-white">
              {subAffiliatesCount} Bots
            </span>
          </div>
          <p className="mt-3.5 text-[11px] font-mono text-gray-400">
            Garantiza <strong className="text-amber-400 font-semibold">+3.92 FGC/h</strong> de comisión de red
          </p>
          <div className="w-full h-[2px] bg-amber-500/20 absolute bottom-0 left-0" />
        </div>

        {/* Metric 4: General Contribution Power / Health */}
        <div className="relative group overflow-hidden bg-gradient-to-br from-black/50 to-emerald-950/10 border border-emerald-500/10 rounded-2xl p-5 hover:border-emerald-500/30 transition-all duration-300">
          <div className="absolute top-2 right-2 w-8 h-8 rounded-lg bg-emerald-500/10 flex items-center justify-center text-emerald-400">
            <Zap className="w-4.5 h-4.5" />
          </div>
          <p className="text-[10px] font-mono font-bold tracking-widest text-emerald-400 uppercase">
            Poder de Aportación
          </p>
          <div className="mt-2.5 flex items-baseline gap-1.5">
            <span className="text-3xl font-display font-extrabold tracking-tight text-white">
              {totalPower}%
            </span>
          </div>
          
          {/* Contribution Health Bar */}
          <div className="mt-3">
            <div className="flex justify-between items-center text-[10px] font-mono text-gray-500 mb-1">
              <span>Salud: Óptima</span>
              <span>Min. 60%</span>
            </div>
            <div className="w-full h-1.5 rounded-full bg-gray-800 overflow-hidden">
              <div 
                className="h-full bg-gradient-to-r from-cyan-400 to-emerald-400 transition-all duration-500" 
                style={{ width: `${totalPower}%` }}
              />
            </div>
          </div>
          <div className="w-full h-[2px] bg-emerald-500/20 absolute bottom-0 left-0" />
        </div>

      </div>

      {/* Hardware Node Contribution Chart & Referral link side-by-side */}
      <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
        
        {/* Modern, glowing vector SVG Line Chart */}
        <div className="cyber-card p-5 rounded-2xl lg:col-span-2 relative">
          <div className="flex justify-between items-center mb-4 pb-2 border-b border-white/5">
            <div className="flex items-center gap-2">
              <BarChart3 className="w-4 h-4 text-[#00d2d3]" />
              <h4 className="font-display font-bold text-sm text-gray-100">
                Gráfico de Rendimiento en Tiempo Real
              </h4>
            </div>
            <div className="flex gap-1.5 bg-black/40 p-1 rounded-lg border border-white/5 text-xs font-mono">
              <button 
                onClick={() => setChartTimeframe('day')}
                className={`px-2 py-1 rounded cursor-pointer transition-colors ${chartTimeframe === 'day' ? 'bg-[#00d2d3]/20 text-[#00d2d3]' : 'text-gray-400 hover:text-white'}`}
              >
                24 Horas
              </button>
              <button 
                onClick={() => setChartTimeframe('week')}
                className={`px-2 py-1 rounded cursor-pointer transition-colors ${chartTimeframe === 'week' ? 'bg-[#00d2d3]/20 text-[#00d2d3]' : 'text-gray-400 hover:text-white'}`}
              >
                Sema.
              </button>
            </div>
          </div>

          {/* Interactive Chart Visual */}
          <div className="relative w-full h-[150px] sm:h-[180px]">
            <svg 
              viewBox={`0 0 ${width} ${height}`} 
              width="100%" 
              height="100%" 
              preserveAspectRatio="none"
              className="overflow-visible"
            >
              <defs>
                {/* Gradient for area color */}
                <linearGradient id="chartGradient" x1="0" y1="0" x2="0" y2="1">
                  <stop offset="0%" stopColor="#00d2d3" stopOpacity="0.25" />
                  <stop offset="100%" stopColor="#6c5ce7" stopOpacity="0" />
                </linearGradient>
                {/* Glow filter */}
                <filter id="glow" x="-20%" y="-20%" width="140%" height="140%">
                  <feDropShadow dx="0" dy="2" stdDeviation="4" floodColor="#00d2d3" floodOpacity="0.4" />
                </filter>
              </defs>

              {/* Grid Lines */}
              <line x1={padding} y1={padding} x2={width - padding} y2={padding} stroke="rgba(255,255,255,0.03)" strokeWidth="1" />
              <line x1={padding} y1={padding + chartHeight/2} x2={width - padding} y2={padding + chartHeight/2} stroke="rgba(255,255,255,0.03)" strokeWidth="1" />
              <line x1={padding} y1={padding + chartHeight} x2={width - padding} y2={padding + chartHeight} stroke="rgba(255,255,255,0.08)" strokeWidth="1" />

              {/* Area filled path */}
              {areaPointsString && (
                <polygon points={areaPointsString} fill="url(#chartGradient)" />
              )}

              {/* Glowing Line */}
              {pointsString && (
                <polyline 
                  fill="none" 
                  stroke="url(#lineGradient)" 
                  strokeWidth="2.5" 
                  points={pointsString} 
                  filter="url(#glow)"
                />
              )}
              {/* Define dynamic line color gradient */}
              <defs>
                <linearGradient id="lineGradient" x1="0" y1="0" x2="1" y2="0">
                  <stop offset="0%" stopColor="#6c5ce7" />
                  <stop offset="50%" stopColor="#00d2d3" />
                  <stop offset="100%" stopColor="#a463f5" />
                </linearGradient>
              </defs>

              {/* Chart data dots */}
              {chartPoints.map((p, index) => {
                const x = padding + (index / (chartPoints.length - 1)) * chartWidth;
                const ratio = maxVal === minVal ? 0.5 : (p.fgc - minVal) / (maxVal - minVal);
                const y = padding + chartHeight - ratio * chartHeight;
                
                // Only render dots for few nodes
                return (
                  <g key={index} className="group/dot">
                    <circle 
                      cx={x} 
                      cy={y} 
                      r={index === chartPoints.length - 1 ? "4.5" : "3"} 
                      className="fill-cyan-300 stroke-[#080816] stroke-1.5 transition-all cursor-pointer group-hover/dot:r-5" 
                    />
                    <text 
                      x={x} 
                      y={height - 5} 
                      fontSize="9" 
                      fill="#888" 
                      fontFamily="monospace"
                      textAnchor="middle"
                    >
                      {p.label}
                    </text>
                  </g>
                );
              })}
            </svg>
            
            {/* Live blinking visual */}
            <div className="absolute top-2 right-4 flex items-center gap-1.5 text-[10px] font-mono text-[#00d2d3] bg-[#00d2d3]/5 px-2 py-0.5 rounded border border-[#00d2d3]/20">
              <span className="w-1.5 h-1.5 rounded-full bg-cyan-400 animate-ping" />
              <span>MINANDO LIVE</span>
            </div>
          </div>
          
          <p className="text-[11px] text-gray-500 font-mono mt-3 text-center sm:text-left flex items-center gap-1">
            <Info className="w-3.5 h-3.5 inline text-cyan-400 shrink-0" />
            <span>Este gráfico muestra el acumulativo en tiempo real de tu nodo Neuraforge y bots de afiliados.</span>
          </p>
        </div>

        {/* Dynamic Referral Card */}
        <div className="cyber-card p-5 rounded-2xl flex flex-col justify-between">
          <div>
            <div className="flex items-center gap-2 mb-3 pb-2 border-b border-white/5">
              <div className="w-6 h-6 rounded bg-amber-500/10 flex items-center justify-center text-amber-400">
                <Users className="w-3.5 h-3.5" />
              </div>
              <h4 className="font-display font-semibold text-sm text-gray-100">
                Tu Enlace de Afiliado
              </h4>
            </div>
            <p className="text-xs text-gray-400 leading-relaxed">
              Invita a amigos, familiares u oficinas a donar o conectar su hardware inactivo a la red NeuraforgeAI.
            </p>
            
            {/* Commission terms list */}
            <ul className="text-[11px] text-gray-300 space-y-2 mt-4 bg-black/30 p-3 rounded-lg border border-white/5">
              <li className="flex justify-between">
                <span className="text-gray-500">Comisión Base:</span>
                <span className="text-[#00d2d3] font-mono">+12% por nodo</span>
              </li>
              <li className="flex justify-between">
                <span className="text-gray-500">Retiro Automático:</span>
                <span className="text-purple-300 font-mono">Soportado BTC</span>
              </li>
              <li className="flex justify-between">
                <span className="text-gray-500">Bono de Bienvenida:</span>
                <span className="text-amber-400 font-mono">5.00 FGC</span>
              </li>
            </ul>
          </div>

          <div className="mt-4">
            <label className="text-[10px] font-mono text-gray-400 uppercase tracking-widest block mb-1.5">
              Enlace Exclusivo de Referido
            </label>
            <div className="flex gap-2 w-full">
              <input 
                type="text" 
                value={referralLink} 
                readonly 
                onClick={(e) => (e.target as HTMLInputElement).select()}
                className="flex-1 px-3 py-2 text-xs font-mono border border-white/10 rounded-xl bg-black/50 text-[#00d2d3] focus:outline-none focus:border-[#00d2d3]/50 truncate select-all"
              />
              <button 
                onClick={handleCopy}
                className={`px-3 py-2 rounded-xl border font-bold text-xs cursor-pointer flex items-center justify-center shrink-0 transition-all ${
                  copied 
                    ? 'bg-emerald-500/30 text-emerald-400 border-emerald-500/30' 
                    : 'bg-purple-600 hover:bg-purple-500 text-white border-transparent hover:shadow-[0_0_15px_rgba(164,99,245,0.35)]'
                }`}
                title="Copiar enlace"
              >
                {copied ? <Check className="w-3.5 h-3.5 shrink-0" /> : <Copy className="w-3.5 h-3.5 shrink-0" />}
              </button>
            </div>
            {copied && (
              <span className="text-[10px] text-emerald-400 font-mono block mt-1 text-right animate-pulse">
                ¡Enlace copiado al portapapeles!
              </span>
            )}
          </div>
        </div>

      </div>

    </div>
  );
}
